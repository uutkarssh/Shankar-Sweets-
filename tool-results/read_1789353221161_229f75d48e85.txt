    55→    55→    55→function DraggablePin({ position, onMove }: { position: [number, number]; onMove: (p: [number, number]) => void }) {
    56→    56→    56→  return (
    57→    57→    57→    <Marker
    58→    58→    58→      position={position}
    59→    59→    59→      icon={pinIcon}
    60→    60→    60→      draggable
    61→    61→    61→      eventHandlers={{
    62→    62→    62→        dragend: (e: any) => {
    63→    63→    63→          const m = e.target as L.Marker
    64→    64→    64→          const ll = m.getLatLng()
    65→    65→    65→          onMove([ll.lat, ll.lng])
    66→    66→    66→        },
    67→    67→    67→        drag: (e: any) => {
    68→    68→    68→          // Live update during drag so the distance label updates in
    69→    69→    69→          // real-time as the user moves the pin.
    70→    70→    70→          const m = e.target as L.Marker
    71→    71→    71→          const ll = m.getLatLng()
    72→    72→    72→          onMove([ll.lat, ll.lng])
    73→    73→    73→        },
    74→    74→    74→      }}
    75→    75→    75→    />
    76→    76→    76→  )
    77→    77→    77→}

   188→   188→   188→      active = false

   208→   208→   208→      goToLogin('location')

   210→   210→   210→    }

   212→   212→   212→      toast.error('Please fill all address fields (house/flat, street, city, pincode)')
   213→   213→   213→      return
   214→   214→   214→    }
   215→   215→   215→    if (!withinRadius) {
   216→   216→   216→      toast.error(`This location is ${distKm.toFixed(2)} km away — outside our ${RESTAURANT.deliveryRadiusKm} km delivery range.`)
   217→   217→   217→      return

   220→   220→   220→    try {
   221→   221→   221→      const res = await authedFetch('/api/addresses', {
   222→   222→   222→        method: 'POST',
   223→   223→   223→        headers: { 'Content-Type': 'application/json' },

   229→   229→   229→          city,

   232→   232→   232→          longitude: pin[1],
   233→   233→   233→          distanceKm: Number(distKm.toFixed(2)),
   234→   234→   234→          isDefault: true,
   235→   235→   235→        }),

   253→   253→   253→    }
