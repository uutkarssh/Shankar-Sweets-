    55→function DraggablePin({ position, onMove }: { position: [number, number]; onMove: (p: [number, number]) => void }) {
    56→  return (
    57→    <Marker
    58→      position={position}
    59→      icon={pinIcon}
    60→      draggable
    61→      eventHandlers={{
    62→        dragend: (e: any) => {
    63→          const m = e.target as L.Marker
    64→          const ll = m.getLatLng()
    65→          onMove([ll.lat, ll.lng])
    66→        },
    67→        drag: (e: any) => {
    68→          // Live update during drag so the distance label updates in
    69→          // real-time as the user moves the pin.
    70→          const m = e.target as L.Marker
    71→          const ll = m.getLatLng()
    72→          onMove([ll.lat, ll.lng])
    73→        },
    74→      }}
    75→    />
    76→  )
    77→}

    79→function ClickToMove({ onMove }: { onMove: (p: [number, number]) => void }) {
    80→  useMapEvents({
    81→    click(e) {
    82→      onMove([e.latlng.lat, e.latlng.lng])
    83→    },
    84→  })
    85→  return null
    86→}

    88→function Recenter({ center }: { center: [number, number] }) {
    89→  const map = useMap()
    90→  useEffect(() => {
    91→    map.flyTo(center, Math.max(map.getZoom(), 15), { duration: 0.6 })
    92→  }, [center[0], center[1]])
    93→  return null
    94→}

   473→function SavedAddresses({ selectedId, onSelect }: { selectedId: string | null; onSelect: (id: string) => void }) {
   474→  const { profile, loading: authLoading } = useAuth()
   475→  const qc = useQueryClient()
   476→  const { data, isLoading } = useQuery({
   477→    queryKey: ['addresses'],
   478→    queryFn: async () => {
   479→      const res = await authedFetch('/api/addresses')
   480→      if (!res.ok) return { addresses: [] }
   481→      return res.json() as Promise<{ addresses: Address[] }>
   482→    },
   483→    enabled: !authLoading && !!profile,
   484→  })
   485→  const [editingId, setEditingId] = useState<string | null>(null)
   486→  const [editLabel, setEditLabel] = useState<'Home' | 'Work' | 'Other'>('Home')
   487→
   488→  if (authLoading) return null
   489→  if (!profile) return null
   490→  const addresses = data?.addresses ?? []
   491→
   492→  if (isLoading && addresses.length === 0) {
   493→    return (
   494→      <section>
   495→        <h3 className="mb-2 text-sm font-bold text-foreground">Saved Addresses</h3>
   496→        <div className="flex flex-col gap-2">
   497→          {Array.from({ length: 2 }).map((_, i) => (
   498→            <div key={i} className="flex items-start gap-2 rounded-xl border border-border/60 bg-card p-3">
   499→              <div className="mt-0.5 h-4 w-4 animate-pulse rounded bg-muted" />
   500→              <div className="flex-1 space-y-1.5">
   501→                <div className="h-3 w-1/3 animate-pulse rounded bg-muted" />
   502→                <div className="h-2.5 w-2/3 animate-pulse rounded bg-muted" />
   503→              </div>
   504→            </div>
   505→          ))}
   506→        </div>
   507→      </section>
   508→    )
   509→  }
   510→  if (addresses.length === 0) return null
   511→
   512→  async function setDefault(id: string) {
   513→    try {
   514→      const res = await authedFetch(`/api/addresses/${id}`, {
   515→        method: 'PATCH',
   516→        headers: { 'Content-Type': 'application/json' },
   517→        body: JSON.stringify({ isDefault: true }),
   518→      })
   519→      if (!res.ok) throw new Error('Failed to set default')
   520→      toast.success('Default address updated')
   521→      qc.invalidateQueries({ queryKey: ['addresses'] })
   522→    } catch (e: any) {
   523→      toast.error(e.message)
   524→    }
   525→  }
   526→
   527→  async function deleteAddress(id: string) {
   528→    if (!confirm('Delete this saved address?')) return
   529→    try {
   530→      const res = await authedFetch(`/api/addresses/${id}`, { method: 'DELETE' })
   531→      if (!res.ok) throw new Error('Failed to delete')
   532→      toast.success('Address deleted')
   533→      qc.invalidateQueries({ queryKey: ['addresses'] })
   534→    } catch (e: any) {
   535→      toast.error(e.message)
   536→    }
   537→  }
   538→
   539→  return (
   540→    <section>
   541→      <h3 className="mb-2 text-sm font-bold text-foreground">Saved Addresses</h3>
   542→      <div className="flex flex-col gap-2">
   543→        {addresses.map((a) => {
   544→          const isSelected = selectedId === a.id
   545→          const isEditing = editingId === a.id
   546→          return (
   547→            <div
   548→              key={a.id}
   549→              className={`rounded-xl border p-3 transition ${
   550→                isSelected ? 'border-brand bg-brand-softer' : 'border-border bg-card'
   551→              }`}
   552→            >
   553→              <button
   554→                onClick={() => !isEditing && onSelect(a.id)}
   555→                className="flex w-full items-start gap-2 text-left"
   556→              >
   557→                <MapPin className="mt-0.5 h-4 w-4 shrink-0 text-brand" />
   558→                <span className="min-w-0 flex-1">
   559→                  <span className="flex items-center gap-2">
   560→                    <span className="text-sm font-bold text-foreground">
   561→                      {a.label ?? 'Address'}
   562→                    </span>
   563→                    {a.isDefault && (
   564→                      <span className="inline-flex items-center gap-0.5 rounded-full bg-brand-softer px-1.5 py-0.5 text-[10px] font-bold text-brand">
   565→                        <Star className="h-2.5 w-2.5 fill-brand" /> Default
   566→                      </span>
   567→                    )}
   568→                  </span>
   569→                  <span className="mt-0.5 block text-xs text-muted-foreground">
   570→                    {a.houseFlat}, {a.streetArea}, {a.city} - {a.pincode}
   571→                  </span>
   572→                  {a.distanceKm != null && (
   573→                    <span className="mt-0.5 block text-[11px] text-muted-foreground">
   574→                      {a.distanceKm.toFixed(2)} km from restaurant
   575→                    </span>
   576→                  )}
   577→                </span>
   578→              </button>
   579→              {/* Edit / Delete / Set-default row */}
   580→              {!isEditing && (
   581→                <div className="mt-2 flex flex-wrap items-center gap-2 pl-6">
   582→                  {!a.isDefault && (
   583→                    <button
   584→                      onClick={() => setDefault(a.id)}
   585→                      className="rounded-md bg-muted px-2 py-1 text-[11px] font-semibold text-foreground hover:bg-muted/70"
   586→                    >
   587→                      Set as default
   588→                    </button>
   589→                  )}
   590→                  <button
   591→                    onClick={() => {
   592→                      setEditingId(a.id)
   593→                      setEditLabel((a.label as 'Home' | 'Work' | 'Other') ?? 'Other')
   594→                    }}
   595→                    className="inline-flex items-center gap-1 rounded-md bg-muted px-2 py-1 text-[11px] font-semibold text-foreground hover:bg-muted/70"
   596→                  >
   597→                    <Pencil className="h-3 w-3" /> Edit
   598→                  </button>
   599→                  <button
   600→                    onClick={() => deleteAddress(a.id)}
   601→                    className="inline-flex items-center gap-1 rounded-md bg-red-50 px-2 py-1 text-[11px] font-semibold text-red-700 hover:bg-red-100"
   602→                  >
   603→                    <Trash2 className="h-3 w-3" /> Delete
   604→                  </button>
   605→                </div>
   606→              )}
   607→              {isEditing && (
   608→                <EditAddressForm
   609→                  address={a}
   610→                  initialLabel={editLabel}
   611→                  onCancel={() => setEditingId(null)}
   612→                  onSaved={() => {
   613→                    setEditingId(null)
   614→                    qc.invalidateQueries({ queryKey: ['addresses'] })
   615→                  }}
   616→                />
   617→              )}
   618→            </div>
   619→          )
   620→        })}
   621→      </div>
   622→    </section>
   623→  )
   624→}
   625→
   626→function EditAddressForm({
   627→  address,
   628→  initialLabel,
   629→  onCancel,
   630→  onSaved,
   631→}: {
   632→  address: Address
   633→  initialLabel: 'Home' | 'Work' | 'Other'
   634→  onCancel: () => void
   635→  onSaved: () => void
   636→}) {
   637→  const [label, setLabel] = useState<'Home' | 'Work' | 'Other'>(initialLabel)
   638→  const [houseFlat, setHouseFlat] = useState(address.houseFlat)
   639→  const [streetArea, setStreetArea] = useState(address.streetArea)
   640→  const [landmark, setLandmark] = useState(address.landmark ?? '')
   641→  const [city, setCity] = useState(address.city)
   642→  const [pincode, setPincode] = useState(address.pincode)
   643→  const [busy, setBusy] = useState(false)
   644→
   645→  async function save() {
   646→    setBusy(true)
   647→    try {
   648→      const res = await authedFetch(`/api/addresses/${address.id}`, {
   649→        method: 'PATCH',
   650→        headers: { 'Content-Type': 'application/json' },
   651→        body: JSON.stringify({ label, houseFlat, streetArea, landmark, city, pincode }),
   652→      })
   653→      if (!res.ok) throw new Error('Failed to update')
   654→      toast.success('Address updated')
   655→      onSaved()
   656→    } catch (e: any) {
   657→      toast.error(e.message)
   658→    } finally {
   659→      setBusy(false)
   660→    }
   661→  }
   662→
   663→  return (
   664→    <div className="mt-3 rounded-lg border border-border bg-muted/30 p-3">
   665→      <div className="mb-2 flex gap-2">
   666→        {(['Home', 'Work', 'Other'] as const).map((l) => (
   667→          <button
   668→            key={l}
   669→            onClick={() => setLabel(l)}
   670→            className={`flex-1 rounded-full px-2 py-1 text-[11px] font-semibold transition ${
   671→              label === l ? 'bg-brand text-brand-foreground' : 'bg-muted text-muted-foreground'
   672→            }`}
   673→          >
   674→            {l}
   675→          </button>
   676→        ))}
   677→      </div>
   678→      <input
   679→        value={houseFlat}
   680→        onChange={(e) => setHouseFlat(e.target.value)}
   681→        placeholder="House / Flat"
   682→        className="mb-2 w-full rounded-md bg-white px-2 py-1.5 text-xs outline-none focus:ring-2 focus:ring-brand/40"
   683→      />
   684→      <input
   685→        value={streetArea}
   686→        onChange={(e) => setStreetArea(e.target.value)}
   687→        placeholder="Street / Area"
   688→        className="mb-2 w-full rounded-md bg-white px-2 py-1.5 text-xs outline-none focus:ring-2 focus:ring-brand/40"
   689→      />
   690→      <input
   691→        value={landmark}
   692→        onChange={(e) => setLandmark(e.target.value)}
   693→        placeholder="Landmark (optional)"
   694→        className="mb-2 w-full rounded-md bg-white px-2 py-1.5 text-xs outline-none focus:ring-2 focus:ring-brand/40"
   695→      />
   696→      <div className="mb-2 grid grid-cols-2 gap-2">
   697→        <input
   698→          value={city}
   699→          onChange={(e) => setCity(e.target.value)}
   700→          placeholder="City"
   701→          className="w-full rounded-md bg-white px-2 py-1.5 text-xs outline-none focus:ring-2 focus:ring-brand/40"
   702→        />
   703→        <input
   704→          value={pincode}
   705→          onChange={(e) => setPincode(e.target.value.replace(/[^0-9]/g, '').slice(0, 6))}
   706→          placeholder="PIN"
   707→          inputMode="numeric"
   708→          className="w-full rounded-md bg-white px-2 py-1.5 text-xs outline-none focus:ring-2 focus:ring-brand/40"
   709→        />
   710→      </div>
   711→      <div className="flex gap-2">
   712→        <button
   713→          onClick={save}
   714→          disabled={busy}
   715→          className="flex-1 rounded-md bg-brand py-1.5 text-xs font-bold text-brand-foreground disabled:opacity-50"
   716→        >
   717→          {busy ? 'Saving…' : 'Save'}
   718→        </button>
   719→        <button
   720→          onClick={onCancel}
   721→          className="flex-1 inline-flex items-center justify-center gap-1 rounded-md border border-border py-1.5 text-xs font-semibold text-foreground"
   722→        >
   723→          <X className="h-3 w-3" /> Cancel
   724→        </button>
   725→      </div>
   726→    </div>
   727→  )
   728→}
